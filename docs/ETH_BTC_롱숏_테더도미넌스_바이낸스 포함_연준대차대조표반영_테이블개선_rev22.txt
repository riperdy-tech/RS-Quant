// =========================

f_ratio_scaled(longClose, shortClose) =>
    r = longClose / math.max(shortClose, 1e-10)
    math.log(math.max(r, 1e-10))

f_macd_hist(src) =>
    macdLine = ta.ema(src, fastLen) - ta.ema(src, slowLen)
    signal   = ta.ema(macdLine, signalLen)
    macdLine - signal

f_scale_generic(x, mode, len) =>
    float out = na
    if mode == "Raw"
        out := x
    else if mode == "Z-score (rolling)"
        mu = ta.sma(x, len)
        sd = ta.stdev(x, len)
        out := (x - mu) / math.max(sd, 1e-10)
    else
        // MinMax (rolling)
        lo = ta.lowest(x, len)
        hi = ta.highest(x, len)
        out := (x - lo) / math.max(hi - lo, 1e-10) * 2.0 - 1.0
    out

f_hist_color(h) =>
    bool up = h >= 0
    bool rising = h > h[1]
    color c = color.gray
    if up and rising
        c := color.lime
    else if up and not rising
        c := color.rgb(211, 199, 63)
    else if (not up) and (not rising)
        c := color.red
    else
        c := color.maroon
    c

// 타임프레임별 최적 설정 함수
f_get_optimal_settings() =>
    int optimal_length = 3
    float optimal_threshold = 0.1
    
    // 현재 타임프레임 분 단위로 변환
    tf_minutes = timeframe.in_seconds() / 60
    
    if tf_minutes <= 1
        // 1분봉 - 유동성은 거의 변화 없으므로 매우 높은 임계값
        optimal_length := 10
        optimal_threshold := 0.3
    else if tf_minutes <= 5
        // 5분봉 이하
        optimal_length := 8
        optimal_threshold := 0.25
    else if tf_minutes <= 15
        // 15분봉
        optimal_length := 7
        optimal_threshold := 0.2
    else if tf_minutes <= 60
        // 1시간봉
        optimal_length := 6
        optimal_threshold := 0.15
    else if tf_minutes <= 120
        // 2시간봉
        optimal_length := 5
        optimal_threshold := 0.12
    else if tf_minutes <= 240
        // 4시간봉
        optimal_length := 4
        optimal_threshold := 0.08
    else if tf_minutes <= 1440
        // 일봉 - Fed 데이터 주 업데이트에 적합
        optimal_length := 3
        optimal_threshold := 0.05
    else if tf_minutes <= 10080
        // 주봉
        optimal_length := 2
        optimal_threshold := 0.03
    else
        // 월봉 이상
        optimal_length := 2
        optimal_threshold := 0.02
    
    [optimal_length, optimal_threshold]

    // 일봉: 3~5일 전과 비교
    liq_lookback_bars := 3

    // 4시간봉 이상: 약 2일에 해당하는 캔들 수
    liq_lookback_bars := math.round(2.0 / tf_in_days)
else if tf_in_days >= 0.04167
    // 1시간봉: 약 2일에 해당하는 캔들 수 (48바)
    liq_lookback_bars := math.round(2.0 / tf_in_days)

    liq_lookback_bars := math.round(2.0 / tf_in_days)

    // 5분봉: 약 2일에 해당하는 캔들 수 (576바)

    liq_lookback_bars := math.round(2.0 / tf_in_days)

bullish_strength = warn_bullish ? (liq_has_meaningful_change ? math.min((usdt_strength_abs + liq_strength_abs) / 2, 100) : usdt_strength_abs * 0.7) : 0

// 강도 레벨 분류 함수
f_strength_level(strength) =>
    string level = ""
    if strength < 20
        level := "약함"
    else if strength < 40
        level := "보통"
    else if strength < 60
        level := "강함"
    else if strength < 80
        level := "매우강함"
    else
        level := "극심"
    level
