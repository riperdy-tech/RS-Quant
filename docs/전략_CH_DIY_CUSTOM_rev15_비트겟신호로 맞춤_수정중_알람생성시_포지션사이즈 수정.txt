// ===== Helpers (moved up to avoid forward-reference issues) =====
round_qty(q, dec) =>
    // qty를 거래소/심볼 수량 규칙에 맞게 내림 처리 + 최소수량(step) 보정
    // - stock/futures: 정수 수량 내림 + 최소 1
    // - 그 외(crypto/forex/cfd 등): 소수점 dec 자리 내림 + 최소 step(10^-dec)
    if na(q) or (q <= 0)
        na
    else
        needsInteger = (syminfo.type == "stock") or (syminfo.type == "futures")
        float factor = math.pow(10.0, math.max(dec, 0))
        float step = needsInteger ? 1.0 : (factor > 1 ? (1.0 / factor) : 1.0)

        float rq = na
        if needsInteger
            rq := math.floor(q)
            rq := rq < 1 ? 1 : rq
        else
            rq := factor > 1 ? math.floor(q * factor) / factor : q
            rq := rq < step ? step : rq

        rq

aitf_meanOfKClosest(value_,target_) => 
    closestDistances = array.new_float(aitf_numberOfClosestValues, 1e10) 
    closestValues    = array.new_float(aitf_numberOfClosestValues, 0.0) 
    for i = 1 to aitf_windowSize 
        value = value_[i] 
        distance = math.abs(target_ - value) 
        maxDistIndex = 0 
        maxDistValue = closestDistances.get(0) 
        for j = 1 to aitf_numberOfClosestValues - 1 
            if array.get(closestDistances, j) > maxDistValue
                maxDistIndex := j
                maxDistValue := array.get(closestDistances, j)
        if distance < maxDistValue 
            array.set(closestDistances, maxDistIndex, distance)
            array.set(closestValues, maxDistIndex, value)
    closestValues.sum() / aitf_numberOfClosestValues

// Defines KNN function to perform classification
aitf_knn(aitf_price) => 
    Pos_count = 0 
    Neg_count = 0 
    min_distance = 10e10 
    nearest_index = 0 
    for j = 1 to 10 
        distance = math.sqrt(math.pow(aitf_price[j] - aitf_price, 2)) 
        if distance < min_distance 
            min_distance := distance
            nearest_index := j
            Neg = aitf_c[nearest_index] > aitf_o[nearest_index] 
            Pos = aitf_c[nearest_index] < aitf_o[nearest_index] 
            if Pos 
                Pos_count += 1
            if Neg 
                Neg_count += 1
    output = Pos_count>Neg_count?1:-1

//Conditional Sampling EMA Function 
Cond_EMA(x, cond, n) =>
    var val = array.new_float(0)
    var ema_val = array.new_float(1)
    if bool(cond)
        array.push(val, x)
        if array.size(val) > 1
            array.remove(val, 0)
        if na(array.get(ema_val, 0))
            array.fill(ema_val, array.get(val, 0))
        array.set(ema_val, 0, (array.get(val, 0) - array.get(ema_val, 0)) * (2 / (n + 1)) + array.get(ema_val, 0))
    EMA = array.get(ema_val, 0)
    EMA

//Conditional Sampling SMA Function
Cond_SMA(x, cond, n) =>
    var vals = array.new_float(0)
    if bool(cond)
        array.push(vals, x)
        if array.size(vals) > n
            array.remove(vals, 0)
    SMA = array.avg(vals)
    SMA

//Standard Deviation Function
Stdev(x, n) =>
    math.sqrt(Cond_SMA(math.pow(x, 2), 1, n) - math.pow(Cond_SMA(x, 1, n), 2))

//Range Size Function
rng_size(x, scale, qty, n) =>
    ATR = Cond_EMA(ta.tr(true), 1, n)
    AC = Cond_EMA(math.abs(x - x[1]), 1, n)
    SD = Stdev(x, n)
    rng_size = scale == "Pips" ? qty * 0.0001 : scale == "Points" ? qty * syminfo.pointvalue : scale == "% of Price" ? close * qty / 100 : scale == "ATR" ? qty * ATR : scale == "Average Change" ? qty * AC : scale == "Standard Deviation" ? qty * SD : scale == "Ticks" ? qty * syminfo.mintick : qty
    rng_size

//Two Type Range Filter Function
rng_filt(h, l, rng_, n, type, smooth, sn, av_rf, av_n) =>
    rng_smooth = Cond_EMA(rng_, 1, sn)
    r = smooth ? rng_smooth : rng_
    var rfilt = array.new_float(2, (h + l) / 2)
    array.set(rfilt, 1, array.get(rfilt, 0))
    if type == "Type 1"
        if h - r > array.get(rfilt, 1)
            array.set(rfilt, 0, h - r)
        if l + r < array.get(rfilt, 1)
            array.set(rfilt, 0, l + r)
    if type == "Type 2"
        if h >= array.get(rfilt, 1) + r
            array.set(rfilt, 0, array.get(rfilt, 1) + math.floor(math.abs(h - array.get(rfilt, 1)) / r) * r)
        if l <= array.get(rfilt, 1) - r
            array.set(rfilt, 0, array.get(rfilt, 1) - math.floor(math.abs(l - array.get(rfilt, 1)) / r) * r)
    rng_filt1 = array.get(rfilt, 0)
    hi_band1 = rng_filt1 + r
    lo_band1 = rng_filt1 - r
    rng_filt2 = Cond_EMA(rng_filt1, rng_filt1 != rng_filt1[1], av_n)
    hi_band2 = Cond_EMA(hi_band1, rng_filt1 != rng_filt1[1], av_n)
    lo_band2 = Cond_EMA(lo_band1, rng_filt1 != rng_filt1[1], av_n)
    rng_filt = av_rf ? rng_filt2 : rng_filt1
    hi_band = av_rf ? hi_band2 : hi_band1
    lo_band = av_rf ? lo_band2 : lo_band1
    [hi_band, lo_band, rng_filt]

    else if type == "HMA"

            ta.hma(source, 2)
        else
            ta.hma(source, length)
    else
        ta.vwma(source, length)

kernel_regression(_rqksrc, _size, _h2) =>
    float _currentWeight = 0.
    float _cumulativeWeight = 0.
    for i = 0 to _size + x_0 by 1
        y = _rqksrc[i]
        w = math.pow(1 + math.pow(i, 2) / (math.pow(_h2, 2) * 2 * r), -r)
        _currentWeight := _currentWeight + y * w
        _cumulativeWeight := _cumulativeWeight + w
        _cumulativeWeight
    _currentWeight / _cumulativeWeight

//FUNCTIONS
//HMA
HMA(_hullsrc, _hull_length) =>
    ta.wma(2 * ta.wma(_hullsrc, _hull_length / 2) - ta.wma(_hullsrc, _hull_length), math.round(math.sqrt(_hull_length)))
    //EHMA    
EHMA(_hullsrc, _hull_length) =>
    ta.ema(2 * ta.ema(_hullsrc, _hull_length / 2) - ta.ema(_hullsrc, _hull_length), math.round(math.sqrt(_hull_length)))
    //THMA    
THMA(_hullsrc, _hull_length) =>
    ta.wma(ta.wma(_hullsrc, _hull_length / 3) * 3 - ta.wma(_hullsrc, _hull_length / 2) - ta.wma(_hullsrc, _hull_length), _hull_length)

//SWITCH
Mode(modeSwitch, hullsrc, len) =>
    modeSwitch == "Hma" ? HMA(hullsrc, len) : modeSwitch == "Ehma" ? EHMA(hullsrc, len) : modeSwitch == "Thma" ? THMA(hullsrc, len / 2) : na

//OUT

detectCrossWithLookback(fast_series, slow_series, lookback) =>
    crossUp = fast_series[lookback] > slow_series[lookback] and fast_series[lookback + 1] <= slow_series[lookback + 1]
    crossDown = fast_series[lookback] < slow_series[lookback] and fast_series[lookback + 1] >= slow_series[lookback + 1]

    if crossUp or crossDown
        noCrosses = true
        for i = 0 to lookback - 1 by 1
            if fast_series[i] > slow_series[i] and fast_series[i + 1] <= slow_series[i + 1] or fast_series[i] < slow_series[i] and fast_series[i + 1] >= slow_series[i + 1]
                noCrosses := false
                break
        if noCrosses
            if crossUp
                1
            else
                -1
        else
            0
    else
        0

    sqz_upperKC = sqz_ma + sqz_rangema * sqz_kcMult
    sqz_lowerKC = sqz_ma - sqz_rangema * sqz_kcMult

    // Squeeze state (원본 계산 유지)
    sqzOn  = (sqz_lowerBB > sqz_lowerKC) and (sqz_upperBB < sqz_upperKC)
    sqzOff = (sqz_lowerBB < sqz_lowerKC) and (sqz_upperBB > sqz_upperKC)
    noSqz  = (not sqzOn) and (not sqzOff)

    // Squeeze Momentum value (LazyBear)
    sqz_mid = ((ta.highest(high, sqz_kcLen) + ta.lowest(low, sqz_kcLen)) / 2 + ta.sma(close, sqz_kcLen)) / 2
    sqz_val = ta.linreg(sqz_src - sqz_mid, sqz_kcLen, 0)
    sqz_prev = nz(sqz_val[1])

    // LazyBear bcolor 규칙: 파랑/초록/빨강/주황
    sqz_isBlue   = (sqz_val > 0)  and (sqz_val > sqz_prev)
    sqz_isGreen  = (sqz_val > 0)  and (sqz_val <= sqz_prev)
    sqz_isRed    = (sqz_val <= 0) and (sqz_val < sqz_prev)
    sqz_isOrange = (sqz_val <= 0) and (sqz_val >= sqz_prev)

    // 요청 규칙:
    // - 주황/파랑일 때만 LONG 가능 (SHORT 제거)
    // - 초록/빨강일 때만 SHORT 가능 (LONG 제거)
    sqz_long_ok := sqz_isOrange or sqz_isBlue
    sqz_short_ok := sqz_isGreen or sqz_isRed

    // 오프셋: 히스토그램 크기가 -오프셋~+오프셋 범위면 LONG/SHORT 라벨 제거
    sqz_inOffset = use_sqz_offset_filter and (sqz_val >= -sqz_hist_offset) and (sqz_val <= sqz_hist_offset)
    if sqz_inOffset
        sqz_long_ok := false
        sqz_short_ok := false

// 4. 최종 라벨 (중복 방지 제거: 조건이 맞으면 연속 신호도 허용)
bool final_long = long_trigger_base and sqz_long_ok
bool final_short = short_trigger_base and sqz_short_ok


// 5. [가장 마지막 추가 필터] 직전 N봉 점수(rounded_score) 평균으로 라벨 제거
//    - 평균 ≥ short_remove_avg_th  => SHORT 제거
//    - 평균 ≤ long_remove_avg_th   => LONG 제거
float avg_prev_score = ta.sma(rounded_score, score_avg_len)[1]
bool score_filter_long_ok  = use_score_avg_filter ? (na(avg_prev_score) ? true : (avg_prev_score > long_remove_avg_th)) : true
bool score_filter_short_ok = use_score_avg_filter ? (na(avg_prev_score) ? true : (avg_prev_score < short_remove_avg_th)) : true

bool final_long_filtered  = final_long  and score_filter_long_ok
bool final_short_filtered = final_short and score_filter_short_ok

// 6. [추가 마지막 필터] 연속 동일 라벨 제거
//    - 마지막으로 출력된 라벨과 같은 방향의 라벨은 반대 라벨이 나오기 전까지 제거
var int last_label_dir = 0  // 1 = LONG, -1 = SHORT, 0 = none

bool long_candidate  = final_long_filtered
bool short_candidate = final_short_filtered

if use_no_consecutive_label_filter
    long_candidate  := final_long_filtered  and last_label_dir != 1
    short_candidate := final_short_filtered and last_label_dir != -1

// [NON-REPAINT] 봉 마감(확정) 후에만 라벨/알림 발생
bool final_long_out  = long_candidate  and barstate.isconfirmed
bool final_short_out = short_candidate and barstate.isconfirmed

// 연속 동일 라벨 제거 필터의 상태 업데이트도 "확정 봉"에서만 반영
if use_no_consecutive_label_filter
    if final_long_out
        last_label_dir := 1
    else if final_short_out
        last_label_dir := -1

// 5. 출력 (라벨 기준 = 알람 기준 100% 동일)
bool long_label_out  = show_final_ls_labels and final_long_out
bool short_label_out = show_final_ls_labels and final_short_out

plotshape(long_label_out,  title="LONG",  location=location.belowbar, color=color.new(color.lime, 0), style=shape.labelup,   text="LONG",  textcolor=color.black, size=size.normal)
plotshape(short_label_out, title="SHORT", location=location.abovebar, color=color.new(color.red, 0),  style=shape.labeldown, text="SHORT", textcolor=color.white, size=size.normal)

alertcondition(long_label_out,  title="LONG Label (Filters)",  message="{{ticker}} LONG 필터링 @ {{close}}")
alertcondition(short_label_out, title="SHORT Label (Filters)", message="{{ticker}} SHORT 필터링 @ {{close}}")

bool labelLongSignal  = enable_long_trades  and final_long_out
bool labelShortSignal = enable_short_trades and final_short_out

if use_signal_edge
    labelLongSignal  := labelLongSignal  and not final_long_out[1]
    labelShortSignal := labelShortSignal and not final_short_out[1]

plotchar(show_debug_signals and labelLongSignal,  title="DBG LABEL Long",  char="L", location=location.top, size=size.tiny)
plotchar(show_debug_signals and labelShortSignal, title="DBG LABEL Short", char="S", location=location.top, size=size.tiny)

// --- Stop price 저장(진입 시점 기준 고정) ---
var float long_stop_at_entry  = na
var float short_stop_at_entry = na

// 포지션이 청산/전환되어 0이 되면 저장값 리셋
if strategy.position_size == 0
    long_stop_at_entry  := na
    short_stop_at_entry := na

    not na(stopPrice) and stopPrice < entryPrice

is_valid_short_stop(entryPrice, stopPrice) =>
    not na(stopPrice) and stopPrice > entryPrice



// ===== Filters (imported from BB_Breakout) =====
// (moved up near the top for UI ordering)
// ===== End Filters =====

// === Combined filter gates ===
allow_long_filters  = arma_long_condt and lvolatilityCondition and HV_long_condt and adx_long_condt and osc_long_cond and ait_long_cond
allow_short_filters = arma_short_condt and svolatilityCondition and HV_short_condt and adx_short_condt and osc_short_cond and ait_short_cond

// --- Entries ---
if labelLongSignal and allow_long_filters and in_date_range
    if close_on_opposite
        strategy.close("Short")
    if strategy.position_size <= 0

            long_stop_at_entry := okStop ? stopL : na
            strategy.entry("Long", strategy.long, qty=qtyL)

// Opposite-signal close for Long does NOT depend on Enable Short.
// - Even when Enable Short = OFF, a valid short signal (final_short_out) can still close an existing Long.
// - Entry into Short positions is still controlled by labelShortSignal (enable_short_trades).
if close_on_opposite and final_short_out and allow_short_filters
    strategy.close("Long")

if labelShortSignal and allow_short_filters and in_date_range
    if strategy.position_size >= 0
        // 진입 시점 기준 stop 고정
        float stopS = short_stop_line
        bool okStop = is_valid_short_stop(close, stopS)
        float qtyS = calc_entry_qty(false, close)

        if not na(qtyS) and qtyS > 0
            short_stop_at_entry := okStop ? stopS : na
            strategy.entry("Short", strategy.short, qty=qtyS)

// =========================================================
// Take Profit & Stop Loss (첨부파일 방식 반영)
// - Take Profit: PERC / ATR / BOTH, (옵션) 트레일링, (옵션) 부분청산(Qty %)
// - Stop Loss  : (기본) 진입 시점 long_stop/short_stop 고정 + (옵션) 트레일링(ON/TP), (옵션) Break Even
//   * Stop Method는 LINE/PERC/ATR/BOTH로 선택 가능 (LINE = 기존 청산기준선)
// =========================================================

// --- helpers ---
var float tpsl_long_entry_price  = na
var float tpsl_short_entry_price = na
var float tpsl_long_openAtr      = na
var float tpsl_short_openAtr     = na

// 포지션이 청산되면 스냅샷 리셋
if strategy.position_size == 0
    tpsl_long_entry_price  := na
    tpsl_short_entry_price := na
    tpsl_long_openAtr      := na
    tpsl_short_openAtr     := na

// Entry 시점(주문 제출 시점) 스냅샷 저장
// (process_orders_on_close=true 이므로, 실제 체결은 봉 마감 기준이며 entryPrice는 close로 간주)
bool tpsl_enterLong  = labelLongSignal  and (strategy.position_size <= 0)
bool tpsl_enterShort = labelShortSignal and (strategy.position_size >= 0)

if tpsl_enterLong
    tpsl_long_entry_price := close
    tpsl_long_openAtr     := ta.atr(tpsl_atrLength)

if tpsl_enterShort
    tpsl_short_entry_price := close
    tpsl_short_openAtr     := ta.atr(tpsl_atrLength)

// =========================================================
// TAKE PROFIT
// (inputs moved up near the top for UI ordering)

tpsl_getLongTP(entryPrice, openAtr) =>
    switch tpsl_longTP_method
        "PERC" => entryPrice * (1 + tpsl_longTP_perc)
        "ATR"  => entryPrice + tpsl_longTP_atrMul * openAtr
        "BOTH" => math.max(entryPrice * (1 + tpsl_longTP_perc), entryPrice + tpsl_longTP_atrMul * openAtr)
        => na

tpsl_getShortTP(entryPrice, openAtr) =>
    switch tpsl_shortTP_method
        "PERC" => entryPrice * (1 - tpsl_shortTP_perc)
        "ATR"  => entryPrice - tpsl_shortTP_atrMul * openAtr
        "BOTH" => math.min(entryPrice * (1 - tpsl_shortTP_perc), entryPrice - tpsl_shortTP_atrMul * openAtr)
        => na

var float tpsl_longTP_price  = na
var float tpsl_shortTP_price = na

// 포지션 진입 시 TP 가격 고정
if tpsl_enterLong
    tpsl_longTP_price := tpsl_getLongTP(tpsl_long_entry_price, nz(tpsl_long_openAtr, ta.atr(tpsl_atrLength)))

if tpsl_enterShort
    tpsl_shortTP_price := tpsl_getShortTP(tpsl_short_entry_price, nz(tpsl_short_openAtr, ta.atr(tpsl_atrLength)))

// 포지션 종료 시 TP 리셋
if strategy.position_size == 0
    tpsl_longTP_price  := na
    tpsl_shortTP_price := na

// TP 도달 플래그 (StopLoss 트레일링이 "TP" 모드일 때 사용)
var bool tpsl_longTP_hit  = false
var bool tpsl_shortTP_hit = false
if strategy.position_size == 0
    tpsl_longTP_hit  := false
    tpsl_shortTP_hit := false
else
    if strategy.position_size > 0 and not na(tpsl_longTP_price)
        tpsl_longTP_hit := tpsl_longTP_hit[1] or (high >= tpsl_longTP_price)
    else
        tpsl_longTP_hit := false
    if strategy.position_size < 0 and not na(tpsl_shortTP_price)
        tpsl_shortTP_hit := tpsl_shortTP_hit[1] or (low <= tpsl_shortTP_price)
    else
        tpsl_shortTP_hit := false

tpsl_longTP_stepTicks  = not na(tpsl_longTP_price)  ? (tpsl_longTP_price  * tpsl_longTP_distancePerc  / syminfo.mintick) : na
tpsl_shortTP_stepTicks = not na(tpsl_shortTP_price) ? (tpsl_shortTP_price * tpsl_shortTP_distancePerc / syminfo.mintick) : na

plot(tpsl_longTP_price,  title="Long Take Profit",  color=color.new(color.fuchsia, 0), linewidth=1, style=plot.style_linebr, offset=1)
plot(tpsl_shortTP_price, title="Short Take Profit", color=color.new(color.fuchsia, 0), linewidth=1, style=plot.style_linebr, offset=1)

// =========================================================
// STOP LOSS
// (inputs moved up near the top for UI ordering)

// baseSrc는 트레일링 계산에 사용할 기준 (close/high/low/entry 등)
tpsl_getLongSL(baseSrc, entryPrice, openAtr, lineStop) =>
    // LINE 선택 시 기존 청산기준선을 사용하되, lineStop이 없으면 PERC 기준으로 fallback
    float computed = switch tpsl_longSL_method
        "LINE" => not na(lineStop) ? lineStop : baseSrc * (1 - tpsl_longSL_perc)
        "PERC" => baseSrc * (1 - tpsl_longSL_perc)
        "ATR"  => baseSrc - tpsl_longSL_atrMul * openAtr
        "BOTH" => math.max(baseSrc * (1 - tpsl_longSL_perc), baseSrc - tpsl_longSL_atrMul * openAtr)
        => na
    // lineStop이 존재하면, 위험을 더 키우지 않도록(더 멀어지지 않도록) 보수적으로 제한
    not na(lineStop) ? math.max(computed, lineStop) : computed

tpsl_getShortSL(baseSrc, entryPrice, openAtr, lineStop) =>
    float computed = switch tpsl_shortSL_method
        "LINE" => not na(lineStop) ? lineStop : baseSrc * (1 + tpsl_shortSL_perc)
        "PERC" => baseSrc * (1 + tpsl_shortSL_perc)
        "ATR"  => baseSrc + tpsl_shortSL_atrMul * openAtr
        "BOTH" => math.min(baseSrc * (1 + tpsl_shortSL_perc), baseSrc + tpsl_shortSL_atrMul * openAtr)
        => na
    not na(lineStop) ? math.min(computed, lineStop) : computed

// trailing 시작 조건
bool tpsl_longTrailOn  = (tpsl_slTrailingMode == "ON") or (tpsl_slTrailingMode == "TP" and tpsl_longTP_hit)
bool tpsl_shortTrailOn = (tpsl_slTrailingMode == "ON") or (tpsl_slTrailingMode == "TP" and tpsl_shortTP_hit)

var float tpsl_longSL_price  = na
var float tpsl_shortSL_price = na

// 진입 시 SL 초기값 고정
if tpsl_enterLong
    // 기존 long_stop_at_entry(라인) 기준이 먼저 저장된 상태이며, 없으면 퍼센트/ATR로 계산됨
    baseSrcL = tpsl_long_entry_price
    initSL = tpsl_getLongSL(baseSrcL, tpsl_long_entry_price, nz(tpsl_long_openAtr, ta.atr(tpsl_atrLength)), long_stop_at_entry)
    tpsl_longSL_price := initSL

if tpsl_enterShort
    baseSrcS = tpsl_short_entry_price
    initSL = tpsl_getShortSL(baseSrcS, tpsl_short_entry_price, nz(tpsl_short_openAtr, ta.atr(tpsl_atrLength)), short_stop_at_entry)
    tpsl_shortSL_price := initSL

// 포지션이 살아있는 동안(옵션) 트레일링로직으로 SL을 조여감
if strategy.position_size > 0
    if tpsl_longTrailOn
        trailBase = tpsl_slTrailSrc == "CLOSE" ? close : high
        newSL = tpsl_getLongSL(trailBase, nz(tpsl_long_entry_price, strategy.position_avg_price), nz(tpsl_long_openAtr, ta.atr(tpsl_atrLength)), long_stop_at_entry)
        // Break-even: TP 도달 이후 SL을 최소 진입가까지 끌어올림
        if tpsl_longBE and tpsl_longTP_hit and not na(tpsl_long_entry_price)
            newSL := math.max(newSL, tpsl_long_entry_price)
        tpsl_longSL_price := math.max(nz(tpsl_longSL_price[1], newSL), newSL)
    else
        // 트레일링 OFF이면 진입 시점 값 유지
        tpsl_longSL_price := nz(tpsl_longSL_price[1])
else if strategy.position_size < 0
    if tpsl_shortTrailOn
        trailBase = tpsl_slTrailSrc == "CLOSE" ? close : low
        newSL = tpsl_getShortSL(trailBase, nz(tpsl_short_entry_price, strategy.position_avg_price), nz(tpsl_short_openAtr, ta.atr(tpsl_atrLength)), short_stop_at_entry)
        if tpsl_shortBE and tpsl_shortTP_hit and not na(tpsl_short_entry_price)
            newSL := math.min(newSL, tpsl_short_entry_price)
        tpsl_shortSL_price := math.min(nz(tpsl_shortSL_price[1], newSL), newSL)
    else
        tpsl_shortSL_price := nz(tpsl_shortSL_price[1])

// 포지션 종료 시 SL 리셋
if strategy.position_size == 0
    tpsl_longSL_price  := na
    tpsl_shortSL_price := na

plot(tpsl_longSL_price,  title="Long Stop Loss",  color=color.new(color.silver, 0), linewidth=1, style=plot.style_linebr, offset=1)
plot(tpsl_shortSL_price, title="Short Stop Loss", color=color.new(color.silver, 0), linewidth=1, style=plot.style_linebr, offset=1)

// =========================================================
// EXIT ORDERS
// =========================================================
// NOTE (BB_Breakout 원본과 동일 동작):
// - TP Qty % = 0 이면 TP/SL(OCO)을 "전체 수량"에 대해 제출합니다(= Profit/Stop Loss 값이 항상 백테스트에 반영됨)
// - TP Qty % > 0 이면 부분청산 TP/SL + 나머지 수량 SL 을 제출합니다
if strategy.position_size > 0
    posQ = math.abs(strategy.position_size)
    tpQ_raw = posQ * (tpsl_longTP_qtyPerc / 100)
    tpQ_rnd = round_qty(tpQ_raw, exchange_decimal)

    // qty%가 0이거나(또는 라운딩 결과가 0) => 전체 수량 TP/SL
    tp_full = (tpsl_longTP_qtyPerc <= 0) or na(tpQ_rnd) or tpQ_rnd <= 0 or tpQ_rnd >= posQ

    if not na(tpsl_longTP_price) and not na(tpsl_longSL_price)
        if tp_full
            // 전체 수량 TP/SL (qty 파라미터를 주지 않으면 전체 수량에 적용됨)
            strategy.exit("Long TP/SL", from_entry="Long", limit = tpsl_longTP_trailingEnabled ? na : tpsl_longTP_price, stop = tpsl_longSL_price, trail_price = tpsl_longTP_trailingEnabled ? tpsl_longTP_price : na, trail_offset = tpsl_longTP_trailingEnabled ? tpsl_longTP_stepTicks : na)
        else
            // 부분청산 TP/SL
            strategy.exit("Long TP/SL", from_entry="Long", qty = tpQ_rnd, limit = tpsl_longTP_trailingEnabled ? na : tpsl_longTP_price, stop = tpsl_longSL_price, trail_price = tpsl_longTP_trailingEnabled ? tpsl_longTP_price : na, trail_offset = tpsl_longTP_trailingEnabled ? tpsl_longTP_stepTicks : na)
            // 나머지 수량 SL
            strategy.exit("Long SL", from_entry="Long", stop = tpsl_longSL_price)
    else if not na(tpsl_longSL_price)
        // TP 가격이 na인 경우라도 SL은 유지
        strategy.exit("Long SL", from_entry="Long", stop = tpsl_longSL_price)

if strategy.position_size < 0
    posQ = math.abs(strategy.position_size)
    tpQ_raw = posQ * (tpsl_shortTP_qtyPerc / 100)
    tpQ_rnd = round_qty(tpQ_raw, exchange_decimal)

    tp_full = (tpsl_shortTP_qtyPerc <= 0) or na(tpQ_rnd) or tpQ_rnd <= 0 or tpQ_rnd >= posQ

    if not na(tpsl_shortTP_price) and not na(tpsl_shortSL_price)
        if tp_full
            strategy.exit("Short TP/SL", from_entry="Short", limit = tpsl_shortTP_trailingEnabled ? na : tpsl_shortTP_price, stop = tpsl_shortSL_price, trail_price = tpsl_shortTP_trailingEnabled ? tpsl_shortTP_price : na, trail_offset = tpsl_shortTP_trailingEnabled ? tpsl_shortTP_stepTicks : na)
        else
            strategy.exit("Short TP/SL", from_entry="Short", qty = tpQ_rnd, limit = tpsl_shortTP_trailingEnabled ? na : tpsl_shortTP_price, stop = tpsl_shortSL_price, trail_price = tpsl_shortTP_trailingEnabled ? tpsl_shortTP_price : na, trail_offset = tpsl_shortTP_trailingEnabled ? tpsl_shortTP_stepTicks : na)
            strategy.exit("Short SL", from_entry="Short", stop = tpsl_shortSL_price)
    else if not na(tpsl_shortSL_price)
        strategy.exit("Short SL", from_entry="Short", stop = tpsl_shortSL_price)

// - Sends: entry/close (and TP/SL closes). Partial TP is locked off in this version.
// - Patch: positionSize now keeps sign (+long / -short) to match {{strategy.position_size}} semantics
// =========================================================
bitget_group = "🟧🟥🟦🟪 Bitget Webhook"
enable_bitget_webhook = input.bool(true, title="Enable Bitget Webhook JSON alert()", group=bitget_group, tooltip="켜면 포지션 수량(strategy.position_size)이 변할 때마다(주문 체결 기준) 비트겟 웹훅 JSON을 alert()로 전송합니다. 알람 생성 시 조건은 'Any alert() function call'을 선택하세요.")

pos_to_str(p) =>
    p > 0 ? "long" : p < 0 ? "short" : "flat"

pos_prev = strategy.position_size[1]
pos_curr = strategy.position_size
pos_delta = pos_curr - pos_prev

if enable_bitget_webhook and barstate.isconfirmed and not na(pos_prev) and (pos_delta != 0)
    string action = pos_delta > 0 ? "buy" : "sell"
    float contracts = math.abs(pos_delta)                 // matches “this fill size” best
    string marketPos = pos_to_str(pos_curr)
    string prevMarketPos = pos_to_str(pos_prev)
    float positionSize = pos_curr                         // ✅ keep sign to match {{strategy.position_size}}

    // Build JSON (strings) to match Bitget webhook format
    string msg = "{\"action\":\"" + action + "\",\"contracts\":\"" + str.tostring(contracts) + "\",\"marketPosition\":\"" + marketPos + "\",\"positionSize\":\"" + str.tostring(positionSize) + "\",\"prevMarketPosition\":\"" + prevMarketPos + "\",\"price\":\"" + str.tostring(close, format.mintick) + "\",\"symbol\":\"" + syminfo.ticker + "\",\"time\":\"" + str.tostring(timenow) + "\"}"
    alert(msg, alert.freq_once_per_bar_close)
